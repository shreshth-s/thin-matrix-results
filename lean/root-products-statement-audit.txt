ThinMatrix.WeightedRootProducts.prepare_value {F q M : ℕ} [NeZero q] (k t card : ℕ)
  (products : Vector (ThinMatrix.BinaryResidualPolynomial.Block F q) M) (i : Fin M) (r : Fin q) :
  ThinMatrix.BinaryCell.view
      (((ThinMatrix.WeightedRootProducts.prepare k t card products).value.outputs.value.get i).sum.value.get r) =
    ∑ d,
      if card ≤ ↑d then
        ThinMatrix.TensorContraction.residualWeight k t ↑d *
          (ThinMatrix.ResidualPolynomialDegree.decode (products.get i)).coeff (↑d - card) r
      else 0

ThinMatrix.ShiftedLowRunFilter.run_filter.{u_1, u_2, u_3} {R : Type u_1} {A : Type u_2} {B : Type u_3} [CommRing R]
  [Fintype A] [Fintype B] (m n t : ℕ) (X : A → Polynomial R) (Y : B → Polynomial R) (dx : A → ℤ) (dy : B → ℤ)
  (K : A → B → Polynomial R) :
  ∑ d ∈ Finset.Icc (-↑m) ↑m,
      ThinMatrix.TensorContraction.weightedLow n t
        (ThinMatrix.TensorContraction.bilinear (fun a => if dx a = d then X a else 0)
          (fun b => if dy b = d then Y b else 0) K) =
    ∑ a,
      ∑ b,
        if dx a = dy b ∧ dx a ∈ Finset.Icc (-↑m) ↑m then
          ThinMatrix.TensorContraction.weightedLow n t (X a * Y b * K a b)
        else 0

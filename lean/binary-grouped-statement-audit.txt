def ThinMatrix.BinaryPreparedGroupedProgram.totalBits : ℕ → ℕ → ℕ → ℕ → ℕ → ℕ :=
fun n r C A M =>
  ThinMatrix.BinaryGroupedPreparation.preparationBits n r C A +
    M * ThinMatrix.BinaryPreparedGroupedProgram.groupBits n r C A
def ThinMatrix.BinaryGroupedPreparation.preparationBits : ℕ → ℕ → ℕ → ℕ → ℕ :=
fun n r C A =>
  2 ^ n *
    (ThinMatrix.CappedSubsetList.count n r *
      (ThinMatrix.BinaryGroupedCoordinates.addressBudget n + C * ThinMatrix.BinaryGroupedPreparation.inputUnit A))
def ThinMatrix.BinaryGroupedPreparation.inputUnit : ℕ → ℕ :=
fun A => 10 * (A.size + 1) ^ 2 + ThinMatrix.BackendScalarBits.backendScalarUnit A.size
def ThinMatrix.BinaryGroupedCoordinates.addressBudget : ℕ → ℕ :=
fun n => n * (2 * ThinMatrix.BackendScalarBits.backendScalarUnit (2 * n + 2)) + 8 * (2 * n + 3) ^ 2 + 16
def ThinMatrix.BinaryPreparedGroupedProgram.groupBits : ℕ → ℕ → ℕ → ℕ → ℕ :=
fun n r C A => ThinMatrix.CappedSubsetList.count n r * (C * ThinMatrix.BackendScalarBits.backendScalarUnit (n + A.size))
def ThinMatrix.BinaryPreparedGroupedProgram.totalControl : ℕ → ℕ → ℕ → ℕ → ℕ :=
fun n r C M =>
  ThinMatrix.BinaryGroupedPreparation.preparationControl n r C +
        M * ThinMatrix.BinaryPreparedGroupedProgram.groupControl n r C +
      52 * 2 ^ n +
    20
def ThinMatrix.BinaryGroupedPreparation.preparationControl : ℕ → ℕ → ℕ → ℕ :=
fun n r C =>
  ThinMatrix.GroupedPreparation.keyBudget n r + ThinMatrix.GroupedPreparation.keyBudget n n +
      2 ^ n * (ThinMatrix.CappedSubsetList.count n r * (28 * n + 40 * C + 58) + 40) +
    22
def ThinMatrix.BinaryPreparedGroupedProgram.groupControl : ℕ → ℕ → ℕ → ℕ :=
fun n r C => ThinMatrix.CappedSubsetList.count n r * (26 * n + 40 * C + 48) + 3 * C + 56

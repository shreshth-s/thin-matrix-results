ThinMatrix.CWExactDecomposition.terms_bilinear (n : ℕ) (x y : ThinMatrix.CWPowerSpecialization.Index n → ℚ)
  (k : ThinMatrix.CWPowerSpecialization.Index n) :
  ∑ s,
      ((ThinMatrix.CWExactDecomposition.terms n s).right k *
          ∑ i, (ThinMatrix.CWExactDecomposition.terms n s).left i * x i) *
        ∑ j, (ThinMatrix.CWExactDecomposition.terms n s).middle j * y j =
    ThinMatrix.FiniteTrilinear.bilinear
      (fun i j k =>
        ↑(ThinMatrix.PolynomialDegeneration.tensorPower (ThinMatrix.CWPolynomialCertificate.target 16) n i j k))
      x y k

  (f : ThinMatrix.CompactBlocks.PhysicalFactorization r) →
    (p : ThinMatrix.BinaryFoldedFactory.Plan f) →
      (block : Fin (ThinMatrix.CompactBlocks.blockCount n h)) →
        ThinMatrix.BinaryFoldedFormation.Inputs n →
          ℕ → ℕ → ℕ → ThinMatrix.GraphWindowPermutations.Prepared f p ↑block :=
fun {n h r} f p block input bandL bandR ell =>
  have length := min ell (p.L - p.m);
  have transformed := ThinMatrix.GraphWindowTransforms.fromInputs f p block input bandL bandR length;
  have out :=
    ThinMatrix.GraphWindowPermutations.execute length transformed.source.windows.value transformed.transformed.value;
  { transformed := transformed,
    permuted := { value := out.value, bitSteps := out.bitSteps, control := out.control + 8 } }

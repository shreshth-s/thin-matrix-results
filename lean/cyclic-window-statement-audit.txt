def ThinMatrix.BinaryWindowPartition.prepare : {k C : ℕ} →
  [NeZero k] →
    ℕ →
      ℕ →
        ℕ →
          ℕ →
            ThinMatrix.BinaryWindowPartition.Source k C →
              ThinMatrix.BinaryGroupedCoordinates.Run (ThinMatrix.BinaryWindowPartition.Prepared k C) :=
fun {k C} [NeZero k] ell cap normalChannels oppositeChannels X =>
  have side := decide (oppositeChannels ≤ normalChannels);
  have rareBudget := cap - max normalChannels oppositeChannels;
  have out := ThinMatrix.BinaryWindowPartition.execute ell rareBudget side X;
  { value := out.value, bitSteps := out.bitSteps, control := out.control + 8 }
def ThinMatrix.BinaryWindowPartition.execute : {k C : ℕ} →
  [NeZero k] →
    ℕ →
      ℕ →
        Bool →
          ThinMatrix.BinaryWindowPartition.Source k C →
            ThinMatrix.BinaryGroupedCoordinates.Run (ThinMatrix.BinaryWindowPartition.Prepared k C) :=
fun {k C} [NeZero k] ell r normal X =>
  have owners :=
    ThinMatrix.BinaryGroupedPreparation.tabulate X fun i => ThinMatrix.BinaryWindowPartition.assign ell r normal ↑i;
  have keys := Vector.ofFn fun s => s;
  have parts :=
    ThinMatrix.BinaryGroupedPreparation.tabulate keys fun s =>
      ThinMatrix.BinaryWindowPartition.buildComponent s owners.value X;
  { value := { owners := owners.value, components := parts.value }, bitSteps := owners.bitSteps + parts.bitSteps,
    control := owners.control + parts.control + 4 * k + 8 }
def ThinMatrix.BinaryWindowPartition.bitBudget : ℕ → ℕ → ℕ → ℕ :=
fun k C A => k * 3 ^ (2 * k) * (C * ThinMatrix.BackendScalarBits.backendScalarUnit A.size)

fun k C ell =>
  3 ^ (2 * k) * (ThinMatrix.BinaryWindowPartition.assignBudget k ell + 32) + 8 +
        k * (3 ^ (2 * k) * (36 * C + 44) + 40) +
      4 * k +
    16

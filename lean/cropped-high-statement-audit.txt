        ℕ →
          ℕ →
            ThinMatrix.BinarySourceFormation.RawStore L q →
              ThinMatrix.BinarySourceFormation.RawStore L q → ThinMatrix.BinaryCroppedHighBoxes.Execution L q :=
fun L m t q cap A X Y =>
  have size := ThinMatrix.BinaryCroppedHighBoxes.setup L A;
  have left := ThinMatrix.BinaryCroppedHighEncoder.execute L m t q cap true X;
  have right := ThinMatrix.BinaryCroppedHighEncoder.execute L m t q cap false Y;
  have out := ThinMatrix.BinaryHighTable.prepare size.value left.table right.table;
  { left := left, right := right, boxes := out.value,

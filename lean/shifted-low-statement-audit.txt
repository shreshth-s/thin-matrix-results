ThinMatrix.GraphShiftedRootFactors.forms_physical {n h r b ell : ℕ}
  {f : ThinMatrix.CompactBlocks.PhysicalFactorization r} {p : ThinMatrix.BinaryFoldedFactory.Plan f}
  (graph : ThinMatrix.GraphPrefixProgram.Prepared f p b ell) (right : Bool)
  (a b✝ : Fin (ThinMatrix.GoodMapEmbedding.goodSlots p.L p.m).card)
  (target : ThinMatrix.BinaryFoldedFormation.BinaryTable n)
  (request : Fin (ThinMatrix.FoldedBackendExecution.windowQuerySize n)) (i : Fin (2 * (p.L - p.m))) (letter : Fin 3) :
  have roots := graph.permuted.transformed.source.roots;
  have targets := ThinMatrix.GraphCompletionRequests.prepare f p roots.source.shape target;
  have labels :=
    (ThinMatrix.GraphCompletionRequests.residual right
        (ThinMatrix.GraphCompletionRequests.positions (roots.keys.value.masks.get a)).value
        (ThinMatrix.GraphCompletionRequests.positions (roots.keys.value.masks.get b✝)).value
        (targets.physical.value.get request)).value;
  ThinMatrix.GraphShiftedRootFactors.forms graph right a b✝ target request i letter =
    if p.L - p.m ≤ ↑i then
      ThinMatrix.WalshEmbedding.ordinaryOpposite (decide (2 ≤ ↑(labels.get i))) (↑(labels.get i) % 2 == 1)
        (ThinMatrix.SourceCoordinates.oppositeEquiv letter)
    else
      ThinMatrix.ShiftedSourceEmbedding.ordinaryNormal (decide (2 ≤ ↑(labels.get i))) (↑(labels.get i) % 2 == 1)
        (ThinMatrix.DistinguishedResidualSupport.normalRemaining letter)
